    13:   --              Everything in front of nl_loop for 4 vertical stripes (lanes). 
    14:   --              Raster rows are stored in a double-buffered row RAM and issued as 
    15:   --              column groups: column j of lanes 0..3 in 4 consecutive slots, 
    16:   --              starting on a fixed phase, so the lane of a slot is (cycle mod 4). 
    17:   -- 
    18:   --              Stages (In = n cycles after issue): 
    19:   --                I0  issue, row RAM / Rx line buffer read 
    20:   --                I2  read data, neighbour ring  -> Rb, Rc, Rd 
    21:   --                I3  D1, D2, Q3 bounds 
    22:   --                I4  Q1, Q2 
    23:   --                I5  sign merge, context row    -> context RAM read 
    24:   --                I6  candidate contexts 
    25:   --                I8  = L, loop input (candidates muxed from the RAM banks) 
    26:   --                I15 = E, loop exit             -> Rx and context write back 
    27:   -- 
    28:   --              Context RAM: one bank per Q3 candidate, address lane*64 + row, 
    29:   --              ctx = 9*row + bank (row = 9*|Q1|+Q2 after sign merge). The write 
    30:   --              of the slot that left the loop at L-1 is forwarded at L; the loop 
    31:   --              forwards the newest one itself. 
    32:   -- 
    33:   --              Per image: clear sweep of the context RAM (and oClr for the A RAMs), 
    34:   --              then H rows. Ws = W/4 must be >= 8 so that Rx is written back 
    35:   --              before the next row reads it. 
    36:   --
