   363:       -- One clock: drive just after an edge, check at the next. 
   364:       procedure step is 
   365:         variable rdy : std_logic_vector(LANES - 1 downto 0); 
   366:       begin 
   367:         if (not vld and fedImg < nFq) then

   371:             gapLeft := rv.RandInt(0, 20); 
   372:           else

   389:             pixIdx /= 2 * CASES(fq(fedImg)).w) then

   393:           vld    := false;

   404:       end procedure step;
