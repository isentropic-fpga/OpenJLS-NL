   384:                    "idle slot: lane state held");

   386:           next; 
   387:         end if; 
   388:    
   389:         -- Reset slot: run mode but iRst = '1'. Outputs quiet, state held.

   392:           sRunHit    <= '0'; 
   393:           sRunCont   <= '0';

   396:           wait for 1 ns;

   404:

   420:           sRunCont <= '0';

   422:         end if;

   431:

   448:           -- A.15 bits are the prefix, A.16 the tail.

   451:             AffirmIfEqual(req, actBuf(lane)(1 to n15), expBuf(lane)(1 to n15), msg.all & ": A.15 bits");

   454:           end if; 
   455:           AffirmIfEqual(req16, actLen(lane), expLen(lane), msg.all & ": bit-length");

   457:             AffirmIfEqual(req16, actBuf(lane)(n15 + 1 to actLen(lane)), expBuf(lane)(n15 + 1 to expLen(lane)), 
   458:                           msg.all & ": A.16 bits"); 
   459:           end if; 
   460:           AffirmIfEqual(reqLane, stIdx(lane), newIdx(lane), msg.all & ": lane RUNindex after run");

   462:    
   463:           ICover(covTerm, (lane, runTerm(lane)));
