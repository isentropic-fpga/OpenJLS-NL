   327:                    "lane " & to_string(s) & ": valid beat has legal byte count"); 
   328:           ICover(covBeat, (s, boolean'pos(nb = BYTES_OUT))); 
   329:           for i in 0 to nb - 1 loop 
   330:             byte     := oWord(OUT_WIDTH - 1 - i * 8 downto OUT_WIDTH - (i + 1) * 8); 
   331:             lastByte := byte; 
   332:             outBytes(s) := outBytes(s) + 1; 
   333:             Check(SB(SB'low + s), bool2bit(oLast = '1' and i = nb - 1) & byte); 
   334:           end loop;

   336:             AffirmIfEqual(req, lastByte, std_logic_vector'(x"D9"), 
   337:                           "lane " & to_string(s) & ": oLast on the FF D9 footer, image " & to_string(imgsDone(s))); 
   338:             AffirmIf(req, outst(s) > 0, "lane " & to_string(s) & ": oLast only for a closed image"); 
   339:             outst(s)    := outst(s) - 1; 
   340:             imgsDone(s) := imgsDone(s) + 1; 
   341:             outBytes(s) := 0; 
   342:           end if; 
   343:         end if; 
   344:    
   345:         s := (s + 1) mod LANES; 
   346:       end procedure step;

   371:       -- Phase of lane l as seen at its ports (header bytes counted from the 
   372:       -- last oLast; a chained image's header queues behind the open one). 
   373:       -- With BYTES_OUT >= HDR_LEN the header leaves in one beat, so the header 
   374:       -- phase is "iStart seen, header not yet out". 
   375:       impure function phase_hit (p : integer) return boolean is 
   376:       begin 
   377:         for l in 0 to LANES - 1 loop 
   378:           case p is 
   379:             when RP_HEADER => 
   380:               if (outst(l) = 0 and outBytes(l) > 0 and outBytes(l) < HDR_LEN) then return true; end if; 
   381:               if (BYTES_OUT >= HDR_LEN and outst(l) = 0 and outBytes(l) = 0 and imgIdx(l) > imgsDone(l) 
   382:                   and (startSent(l) or not firstWord(l))) then 
   383:                 return true; 
   384:               end if; 
   385:             when RP_DATA   => if (outst(l) = 0 and outBytes(l) >= HDR_LEN and wordsLeft(l) > 0) then return true; end if; 
   386:             when RP_EOI    => if (outst(l) > 0) then return true; end if; 
   387:             when RP_HOLD   => if (held(l)) then return true; end if; 
   388:             when others    => null; 
   389:           end case; 
   390:         end loop; 
   391:         return p = RP_IDLE and all_idle; 
   392:       end function phase_hit;

   394:       -- Reset now: aborted images' expected bytes are dropped, every lane

   407:           outst(l)     := 0; 
   408:           outBytes(l)  := 0; 
   409:           held(l)      := false; 
   410:           chain(l)     := false; 
   411:           notRdy(l)    := 0;

   413:           imgEnd(l)    := imgIdx(l) + 1; 
   414:         end loop; 
   415:       end procedure abort_and_reset;
