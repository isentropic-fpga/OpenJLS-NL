    11:   -- Description: 
    12:   -- 
    13:   --   NL: OpenJLS byte_stuffer shared by 4 lanes taking turns, one lane per 
    14:   --   clock (lane = iSlot). Every register is kept 4 times in a ring X_r(0..3) 
    15:   --   that rotates each clock: slot 0 holds the lane served this cycle, the 
    16:   --   logic reads X = X_r(0) and writes X_r(3). The logic itself is unchanged 
    17:   --   (lines marked NL). The two internal FIFOs become one FIFO per lane, 
    18:   --   written and read only on that lane's turn. All ports belong to lane 
    19:   --   iSlot this cycle, except oAlmostFull (one bit per lane). 
    20:   -- 
    21:   --   Per T.87: every 0xFF byte in the encoded bitstream must be followed by a 
    22:   --   stuffed '0' bit so decoders can distinguish payload from markers (an FF 
    23:   --   followed by a byte with MSB='1' denotes a marker). 
    24:   -- 
    25:   --   Three internal stages: 
    26:   -- 
    27:   --     Stage 1 — bit packer: 
    28:   --       Accumulates input bits MSB-first into a 2*FIFO_BITS accumulator. 
    29:   --       Drains a fixed FIFO_BITS-wide word into the FIFO whenever enough 
    30:   --       bits are present. On flush the sub-byte residue is padded to a 
    31:   --       byte boundary; the final FIFO write is always FIFO_BITS wide 
    32:   --       (zero-padded if needed) and carries the final word's real valid-bit 
    33:   --       count (the byte-boundary pad excluded) plus last_flag=1. Stage 3 
    34:   --       counts only those real bits, so the pad is never emitted; the 
    35:   --       genuine residue is padded post-stuffing at the terminal beat. 
    36:   -- 
    37:   --     Stage 2 — BRAM-backed sync FIFO 
    38:   -- 
    39:   --     Stage 3 — FF stuffer + output emit: 
    40:   --       Refills a holding register using the nine fixed alignments allowed 
    41:   --       by the refill contract (0..8 old bits). Each cycle forms up to four 
    42:   --       output bytes by decoding all eight legal stuffing layouts in 
    43:   --       parallel. Each layout uses fixed bit slices and constant valid-bit 
    44:   --       thresholds. One-hot selection chooses the output, the last FF state, 
    45:   --       and a fixed-shift remainder without a serial lane-resolution chain. 
    46:   -- 
    47:   --       The end-of-image terminal beat (sub-byte residue, a pending stuff 
    48:   --       bit with no follow-up data, or a byte-aligned clean end) is split 
    49:   --       into its own cycle via sLastPending: the final byte (or 0-byte beat) 
    50:   --       is assembled, latched, and emitted on the following beat. Adds at 
    51:   --       most 1 cycle of latency per image boundary and keeps the pad-byte 
    52:   --       assembly off the critical path. 
    53:   -- 
    54:   --   Flush protocol (iFlush, single-cycle pulse from upstream on the cycle 
    55:   --   the bit_packer presents the image's last word): 
    56:   --     - Stage 1 pads its sub-byte residue and tags the final FIFO write 
    57:   --       with last_flag=1. 
    58:   --     - Stage 3 latches the last_flag when it pops that word. Once the 
    59:   --       holding register drains, it inserts a final stuff '0' if the last 
    60:   --       payload byte was 0xFF, zero-pads the output accumulator to a byte 
    61:   --       boundary, emits the remaining whole bytes, and pulses oFlushDone 
    62:   --       on the final output beat (oFlushDone is sampled together with 
    63:   --       oWordValid='1', matching jls_framer's iEOI contract). 
    64:   -- 
    65:   -- Generics: 
    66:   --   IN_WIDTH            : bit_packer worst-case word width (= LIMIT). 
    67:   --   OUT_BYTES_PER_CYCLE : output bytes/cycle; must be 4 (fixed layout table).

   608:     sSkidData <= sSkidWord(FIFO_WIDTH - 1 downto DATA_LSB); 
   609:     sSkidLast <= sSkidWord(LAST_POS); 
   610:    
   611:     skid_proc : process (iClk) is 
   612:     begin 
   613:    
   614:       if rising_edge(iClk) then 
   615:         -- NL: rotate lane rings 
   616:         sSkidValid_r <= sSkidValid_r(1 to 3) & sSkidValid_r(0); 
   617:         sSkidWord_r <= sSkidWord_r(1 to 3) & sSkidWord_r(0); 
   618:         if (iRst = '1') then 
   619:           sSkidValid_r <= (others => '0'); 
   620:           sSkidWord_r <= (others => (others => '0')); 
   621:         else 
   622:           if (sSkidTaken = '1') then 
   623:             sSkidValid_r(3) <= '0'; 
   624:           end if; 
   625:           if (sFifoOutValid = '1' and sFifoOutReady = '1') then 
   626:             sSkidWord_r(3) <= sFifoOutData; 
   627:             sSkidValid_r(3) <= '1'; 
   628:           end if; 
   629:         end if; 
   630:       end if; 
   631:    
   632:     end process skid_proc; 
   633:    
   634:     stage3_proc : process (iClk) is 
   635:    
   636:       variable vStuffBuffer     : std_logic_vector(HOLD_BITS - 1 downto 0); 
   637:       variable vStuffBufferBits : natural range 0 to HOLD_BITS; 
   638:       variable vStuffBufferLast : std_logic; 
   639:       variable vPrevFF          : std_logic; 
   640:    
   641:       variable vValidBitsInt  : natural range 0 to FIFO_BITS; 
   642:       variable vRefillBuffer  : std_logic_vector(HOLD_BITS - 1 downto 0); 
   643:       variable vRefillWord    : std_logic_vector(HOLD_BITS - 1 downto 0); 
   644:       variable vRefillMask    : std_logic_vector(HOLD_BITS - 1 downto 0); 
   645:    
   646:       -- Fixed-window FF flags used to decode the layout. The final emitted 
   647:       -- lane's FF state is computed independently in the candidate loop. 
   648:       variable ff0        : std_logic; -- offset 0 
   649:       variable ff1a, ff1b : std_logic; -- offsets 7, 8 
   650:       variable ff2a, ff2b : std_logic; -- offsets 15, 16 
   651:    
   652:       variable vPath          : std_logic_vector(0 to 7); 
   653:       variable vTakeShift     : std_logic_vector(CONSUME_BITS'range); 
   654:       variable vPathMask      : std_logic_vector(OUT_WIDTH - 1 downto 0); 
   655:       variable vCandidateWord : std_logic_vector(OUT_WIDTH - 1 downto 0); 
   656:       variable vCandidateByte : std_logic_vector(7 downto 0); 
   657:       variable vLaneSelected  : std_logic; 
   658:       variable vBytesMask     : unsigned(sOutBytesValidReg'range); 
   659:       variable vEmitBytesBits : unsigned(sOutBytesValidReg'range); 
   660:       variable vBufferMask    : std_logic_vector(HOLD_BITS - 1 downto 0); 
   661:       variable vNextBuffer    : std_logic_vector(HOLD_BITS - 1 downto 0); 
   662:       variable vCountMask     : unsigned(sStuffBufferBits'range); 
   663:       variable vNextCount     : unsigned(sStuffBufferBits'range); 
   664:    
   665:       variable vEmitData   : std_logic_vector(OUT_WIDTH - 1 downto 0); 
   666:       variable vEmitBytes  : natural range 0 to OUT_BYTES_PER_CYCLE; 
   667:       variable vEmitLastFF : std_logic; 
   668:       variable vPadByte    : std_logic_vector(7 downto 0); 
   669:    
   670:     begin 
   671:    
   672:       if rising_edge(iClk) then 
   673:         -- NL: rotate lane rings 
   674:         sStuffBuffer_r <= sStuffBuffer_r(1 to 3) & sStuffBuffer_r(0); 
   675:         sStuffBufferBits_r <= sStuffBufferBits_r(1 to 3) & sStuffBufferBits_r(0); 
   676:         sStuffBufferLast_r <= sStuffBufferLast_r(1 to 3) & sStuffBufferLast_r(0); 
   677:         sPrevFF_r <= sPrevFF_r(1 to 3) & sPrevFF_r(0); 
   678:         sOutWordReg_r <= sOutWordReg_r(1 to 3) & sOutWordReg_r(0); 
   679:         sOutValidReg_r <= sOutValidReg_r(1 to 3) & sOutValidReg_r(0); 
   680:         sOutBytesValidReg_r <= sOutBytesValidReg_r(1 to 3) & sOutBytesValidReg_r(0); 
   681:         sFlushDone_r <= sFlushDone_r(1 to 3) & sFlushDone_r(0); 
   682:         sLastPending_r <= sLastPending_r(1 to 3) & sLastPending_r(0); 
   683:         if (iRst = '1') then 
   684:           sStuffBuffer_r <= (others => (others => '0')); 
   685:           sStuffBufferBits_r <= (others => (others => '0')); 
   686:           sStuffBufferLast_r <= (others => '0'); 
   687:           sPrevFF_r <= (others => '0'); 
   688:           sOutWordReg_r <= (others => (others => '0')); 
   689:           sOutValidReg_r <= (others => '0'); 
   690:           sOutBytesValidReg_r <= (others => (others => '0')); 
   691:           sFlushDone_r <= (others => '0'); 
   692:           sLastPending_r <= (others => '0'); 
   693:         elsif (sLastPending = '1') then 
   694:           -- EOI terminal beat, assembled outside the layout decoder (1 extra cycle, 
   695:           -- absorbed by the stage 2 FIFO). Sub-byte residue or dangling 0xFF 
   696:           -- emits one padded byte; a byte-aligned clean end emits a 0-byte beat. 
   697:    
   698:           if (iReady = '1') then 
   699:             vStuffBufferBits := to_integer(sStuffBufferBits); 
   700:             vPadByte         := (others => '0'); 
   701:    
   702:             if (sPrevFF = '1') then 
   703:               -- Stuff '0' at MSB, up to 7 real bits below it, zero pad. 
   704:               if (vStuffBufferBits > 0) then 
   705:                 vPadByte(6 downto 7 - vStuffBufferBits) := sStuffBuffer(HOLD_BITS - 1 downto HOLD_BITS - vStuffBufferBits); 
   706:               end if; 
   707:             elsif (vStuffBufferBits > 0) then 
   708:               vPadByte(7 downto 8 - vStuffBufferBits) := sStuffBuffer(HOLD_BITS - 1 downto HOLD_BITS - vStuffBufferBits); 
   709:             end if; 
   710:    
   711:             if (vStuffBufferBits = 0 and sPrevFF = '0') then 
   712:               sOutWordReg_r(3) <= (others => '0'); 
   713:               sOutBytesValidReg_r(3) <= (others => '0'); 
   714:             else 
   715:               sOutWordReg_r(3)(OUT_WIDTH - 1 downto OUT_WIDTH - 8) <= vPadByte; 
   716:               sOutWordReg_r(3)(OUT_WIDTH - 9 downto 0) <= (others => '0'); 
   717:               sOutBytesValidReg_r(3) <= to_unsigned(1, sOutBytesValidReg'length); 
   718:             end if; 
   719:    
   720:             sOutValidReg_r(3) <= '1'; 
   721:             sFlushDone_r(3) <= '1'; 
   722:             sLastPending_r(3) <= '0'; 
   723:             sStuffBufferLast_r(3) <= '0'; 
   724:             sPrevFF_r(3) <= '0'; 
   725:             sStuffBuffer_r(3) <= (others => '0'); 
   726:             sStuffBufferBits_r(3) <= (others => '0'); 
   727:           else 
   728:             sOutValidReg_r(3) <= '0'; 
   729:             sFlushDone_r(3) <= '0'; 
   730:           end if; 
   731:         else 
   732:           vStuffBuffer     := sStuffBuffer; 
   733:           vStuffBufferBits := to_integer(sStuffBufferBits); 
   734:           vStuffBufferLast := sStuffBufferLast; 
   735:           vPrevFF          := sPrevFF; 
   736:           vEmitBytes       := 0; 
   737:           vEmitData        := (others => '0'); 
   738:           sFlushDone_r(3) <= '0'; 
   739:    
   740:           ---------------------------------------------------------------------- 
   741:           -- (1) Refill: drain the skid buffer into the holding buffer. 
   742:           -- Only the final word may be partial. 
   743:           ---------------------------------------------------------------------- 
   744:           if (sSkidTaken = '1') then 
   745:             -- The refill contract allows at most eight old bits. Express the 
   746:             -- nine legal alignments as fixed wiring, including a final FIFO word. 
   747:             -- Invalid trailing bits may be copied: the real-bit count below 
   748:             -- prevents their emission, and the next refill overwrites them. 
   749:             vRefillBuffer := (others => '0'); 
   750:             for offset in 0 to HOLD_BITS - FIFO_BITS loop 
   751:    
   752:               vRefillWord := (others => '0'); 
   753:               if (offset > 0) then 
   754:                 vRefillWord(HOLD_BITS - 1 downto HOLD_BITS - offset) := 
   755:                   sStuffBuffer(HOLD_BITS - 1 downto HOLD_BITS - offset); 
   756:               end if; 
   757:               vRefillWord(HOLD_BITS - 1 - offset downto HOLD_BITS - offset - FIFO_BITS) := sSkidData; 
   758:               vRefillMask := (others => bool2bit(vStuffBufferBits = offset)); 
   759:               vRefillBuffer := vRefillBuffer or (vRefillWord and vRefillMask); 
   760:    
   761:             end loop; 
   762:             vStuffBuffer := vRefillBuffer; 
   763:             if (sSkidLast = '0') then 
   764:               vValidBitsInt := FIFO_BITS; 
   765:             else 
   766:               vValidBitsInt := to_integer(unsigned(sBvQueueOutData)); 
   767:               vStuffBufferLast := '1'; 
   768:             end if; 
   769:             vStuffBufferBits := vStuffBufferBits + vValidBitsInt; 
   770:           end if; 
   771:    
   772:           ---------------------------------------------------------------------- 
   773:           -- (2) Parallel-precompute the layout decoder's fixed-window FF flags. 
   774:           ---------------------------------------------------------------------- 
   775:           ff0  := bool2bit(vStuffBuffer(HOLD_BITS - 1 downto HOLD_BITS - 8) = x"FF"); 
   776:           ff1a := bool2bit(vStuffBuffer(HOLD_BITS - 8 downto HOLD_BITS - 15) = x"FF"); 
   777:           ff1b := bool2bit(vStuffBuffer(HOLD_BITS - 9 downto HOLD_BITS - 16) = x"FF"); 
   778:           ff2a := bool2bit(vStuffBuffer(HOLD_BITS - 16 downto HOLD_BITS - 23) = x"FF"); 
   779:           ff2b := bool2bit(vStuffBuffer(HOLD_BITS - 17 downto HOLD_BITS - 24) = x"FF"); 
   780:    
   781:           ---------------------------------------------------------------------- 
   782:           -- (3) Decode all layouts in parallel. Each predicate depends only on 
   783:           --     the incoming FF state and fixed-window comparisons, never on 
   784:           --     a byte or consumption count selected by another lane. 
   785:           ---------------------------------------------------------------------- 
   786:           vPath(0) := not vPrevFF and not ff0 and not ff1b and not ff2b; 
   787:           vPath(1) := not vPrevFF and not ff0 and not ff1b and     ff2b; 
   788:           vPath(2) := not vPrevFF and not ff0 and     ff1b; 
   789:           vPath(3) := not vPrevFF and     ff0 and not ff2a; 
   790:           vPath(4) := not vPrevFF and     ff0 and     ff2a; 
   791:           vPath(5) :=     vPrevFF and not ff1a and not ff2a; 
   792:           vPath(6) :=     vPrevFF and not ff1a and     ff2a; 
   793:           vPath(7) :=     vPrevFF and     ff1a; 
   794:    
   795:           vEmitData      := (others => '0'); 
   796:           vEmitBytesBits := (others => '0'); 
   797:           vEmitLastFF    := '0'; 
   798:           vTakeShift     := (others => '0'); 
   799:    
   800:           for p in STUFF_LAYOUT'range loop 
   801:    
   802:             vCandidateWord := (others => '0'); 
   803:             vPathMask      := (others => vPath(p)); 
   804:    
   805:             for lane in 0 to OUT_BYTES_PER_CYCLE - 1 loop 
   806:    
   807:               if (STUFF_LAYOUT(p)(lane) = '1') then 
   808:                 vCandidateByte := '0' & vStuffBuffer(HOLD_BITS - 1 - LANE_START(p, lane) 
   809:                                                      downto HOLD_BITS - LANE_END(p, lane)); 
   810:               else 
   811:                 vCandidateByte := vStuffBuffer(HOLD_BITS - 1 - LANE_START(p, lane) 
   812:                                               downto HOLD_BITS - LANE_END(p, lane)); 
   813:               end if; 
   814:               vCandidateWord(OUT_WIDTH - 1 - lane * 8 downto OUT_WIDTH - (lane + 1) * 8) := vCandidateByte; 
   815:    
   816:               -- Select the last complete lane using constant thresholds. The 
   817:               -- end thresholds increase strictly, so exactly one lane wins. 
   818:               vLaneSelected := vPath(p) and iReady and 
   819:                                bool2bit(vStuffBufferBits >= LANE_END(p, lane)); 
   820:               if (lane < OUT_BYTES_PER_CYCLE - 1) then 
   821:                 vLaneSelected := vLaneSelected and 
   822:                                  bool2bit(vStuffBufferBits < LANE_END(p, lane + 1)); 
   823:               end if; 
   824:               vBytesMask     := (others => vLaneSelected); 
   825:               vEmitBytesBits := vEmitBytesBits or (to_unsigned(lane + 1, vEmitBytesBits'length) and vBytesMask); 
   826:               vEmitLastFF    := vEmitLastFF or (vLaneSelected and bool2bit(vCandidateByte = x"FF")); 
   827:    
   828:               for k in CONSUME_BITS'range loop 
   829:    
   830:                 if (CONSUME_BITS(k) = LANE_END(p, lane)) then 
   831:                   vTakeShift(k) := vTakeShift(k) or vLaneSelected; 
   832:                 end if; 
   833:    
   834:               end loop; 
   835:    
   836:             end loop; 
   837:    
   838:             vEmitData := vEmitData or (vCandidateWord and vPathMask); 
   839:    
   840:           end loop; 
   841:    
   842:           ---------------------------------------------------------------------- 
   843:           -- (4) Select fixed shifts and constant count decrements in parallel. 
   844:           --     No general barrel shifter follows a late selected integer count. 
   845:           --     Include the no-emission case so stalls preserve all input bits. 
   846:           ---------------------------------------------------------------------- 
   847:           vEmitBytes := to_integer(vEmitBytesBits); 
   848:           -- No complete first byte: this decision does not depend on the 
   849:           -- layout decoder or the selected output count. 
   850:           vTakeShift(0) := not iReady or bool2bit(vStuffBufferBits < 7) or 
   851:                            (not vPrevFF and bool2bit(vStuffBufferBits = 7)); 
   852:           vNextBuffer := (others => '0'); 
   853:           vNextCount  := (others => '0'); 
   854:    
   855:           for k in CONSUME_BITS'range loop 
   856:    
   857:             vBufferMask := (others => vTakeShift(k)); 
   858:             vCountMask  := (others => vTakeShift(k)); 
   859:             vNextBuffer := vNextBuffer or 
   860:                            (std_logic_vector(shift_left(unsigned(vStuffBuffer), CONSUME_BITS(k))) and vBufferMask); 
   861:             -- Unsigned arithmetic deliberately wraps on unselected candidates 
   862:             -- with too few bits. Their mask is zero; only a legal count wins. 
   863:             vNextCount := vNextCount or 
   864:                           ((to_unsigned(vStuffBufferBits, vNextCount'length) - CONSUME_BITS(k)) and vCountMask); 
   865:    
   866:           end loop; 
   867:    
   868:           vStuffBuffer     := vNextBuffer; 
   869:           vStuffBufferBits := to_integer(vNextCount); 
   870:           if (vEmitBytes > 0) then 
   871:             vPrevFF := vEmitLastFF; 
   872:           end if; 
   873:    
   874:           ---------------------------------------------------------------------- 
   875:           -- (5) Output register and flush-done / drain entry. 
   876:           ---------------------------------------------------------------------- 
   877:           if (vEmitBytes > 0) then 
   878:             sOutWordReg_r(3) <= vEmitData; 
   879:             sOutBytesValidReg_r(3) <= to_unsigned(vEmitBytes, sOutBytesValidReg'length); 
   880:             sOutValidReg_r(3) <= '1'; 
   881:           else 
   882:             sOutValidReg_r(3) <= '0'; 
   883:           end if; 
   884:    
   885:           -- Once the last word is consumed and only a sub-byte residue remains 
   886:           -- (bits < 8, including the bits=0 clean end), hand off to the 
   887:           -- sLastPending branch which assembles the final beat off this path. 
   888:           if (iReady = '1' 
   889:               and vStuffBufferLast = '1' 
   890:               and vStuffBufferBits < 8) then 
   891:             sLastPending_r(3) <= '1'; 
   892:           end if; 
   893:    
   894:           sStuffBuffer_r(3) <= vStuffBuffer; 
   895:           sStuffBufferBits_r(3) <= to_unsigned(vStuffBufferBits, sStuffBufferBits'length); 
   896:           sStuffBufferLast_r(3) <= vStuffBufferLast; 
   897:           sPrevFF_r(3) <= vPrevFF; 
   898:         end if; 
   899:       end if; 
   900:    
   901:     end process stage3_proc; 
   902:    
   903:   end architecture behavioral;
